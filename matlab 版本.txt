%% Rolling Ball 背景减除
function background = rolling_ball_background(img, ball_radius)
    % 使用形态学开运算近似Rolling Ball算法
    se = strel('disk', ball_radius);
    background = imopen(img, se);
end

%% 自定义Colormap：白色+Jet（平滑过渡）
function cmap = custom_jet_white_smooth(n, white_ratio, transition_ratio)
    if nargin < 1, n = 256; end
    if nargin < 2, white_ratio = 0.05; end
    if nargin < 3, transition_ratio = 0.10; end
    
    white_n = round(n * white_ratio);
    transition_n = round(n * transition_ratio);
    jet_n = n - white_n - transition_n;
    
    % 纯白色部分
    white_part = ones(white_n, 3);
    
    % 获取jet的颜色
    jet_colors = jet(max(jet_n, 1));
    if jet_n < 1
        jet_colors = jet(1);
    end
    first_jet_color = jet_colors(1,:);
    
    % 过渡部分：白色渐变到jet的起始色
    transition_part = zeros(transition_n, 3);
    for i = 1:transition_n
        ratio = i / transition_n;
        transition_part(i,:) = (1-ratio) * [1 1 1] + ratio * first_jet_color;
    end
    
    % 合并
    cmap = [white_part; transition_part; jet_colors];
end

%% 自适应中值滤波
function output = adaptive_median_filter(img, Smax)
    [m, n] = size(img);
    output = img;
    
    for i = 1:m
        for j = 1:n
            Sxy = 3;
            while Sxy <= Smax
                half = floor(Sxy/2);
                rmin = max(1, i-half);
                rmax = min(m, i+half);
                cmin = max(1, j-half);
                cmax = min(n, j+half);
                
                window = img(rmin:rmax, cmin:cmax);
                zmin = min(window(:));
                zmax = max(window(:));
                zmed = median(window(:));
                zxy = img(i,j);
                
                A1 = zmed - zmin;
                A2 = zmed - zmax;
                
                if A1 > 0 && A2 < 0
                    B1 = zxy - zmin;
                    B2 = zxy - zmax;
                    if B1 > 0 && B2 < 0
                        output(i,j) = zxy;
                    else
                        output(i,j) = zmed;
                    end
                    break;
                else
                    Sxy = Sxy + 2;
                end
                
                if Sxy > Smax
                    output(i,j) = zmed;
                end
            end
        end
    end
end

%% 开关中值滤波
function output = switching_median_filter(img, window_size)
    [m, n] = size(img);
    output = img;
    half = floor(window_size/2);
    
    for i = half+1:m-half
        for j = half+1:n-half
            window = img(i-half:i+half, j-half:j+half);
            center = img(i,j);
            zmin = min(window(:));
            zmax = max(window(:));
            
            % 如果是极值点，认为是噪声
            if center == zmin || center == zmax
                output(i,j) = median(window(:));
            end
        end
    end
end
